cmake_minimum_required(VERSION 3.20)
project(calibration_capture LANGUAGES CXX)
find_package(Threads REQUIRED)
set(CAMERA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/camera" CACHE PATH "Frozen lower-level RAW capture and luma source")
add_executable(calibration-capture CalibrationCapture.cpp "${CAMERA_SOURCE}/RawOpticalCapture.cpp" "${CAMERA_SOURCE}/BayerLuma.cpp")
target_compile_features(calibration-capture PRIVATE cxx_std_23)
target_compile_options(calibration-capture PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(calibration-capture PRIVATE "${CAMERA_SOURCE}")
target_link_libraries(calibration-capture PRIVATE Threads::Threads)
