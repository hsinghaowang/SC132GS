cmake_minimum_required(VERSION 3.16)
project(ros_replay_receiver LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(ov_msckf REQUIRED)
find_package(ov_core REQUIRED)
find_package(ov_init REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(message_filters REQUIRED)
find_package(image_transport REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED)
find_package(Boost REQUIRED COMPONENTS filesystem system thread date_time)
find_package(Ceres REQUIRED)
add_executable(ros_replay_receiver Receiver.cpp)
target_compile_features(ros_replay_receiver PRIVATE cxx_std_17)
target_compile_definitions(ros_replay_receiver PRIVATE ROS_AVAILABLE=2 ENABLE_ARUCO_TAGS=0)
target_include_directories(ros_replay_receiver PRIVATE ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(ros_replay_receiver ov_msckf ov_core ov_init rclcpp sensor_msgs geometry_msgs nav_msgs message_filters image_transport cv_bridge tf2_ros tf2_geometry_msgs)
target_link_libraries(ros_replay_receiver Eigen3::Eigen ${OpenCV_LIBRARIES} ${Boost_LIBRARIES} Ceres::ceres)
