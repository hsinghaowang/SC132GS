cmake_minimum_required(VERSION 3.16)
project(native_replay_publisher LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgcodecs)
add_executable(native_replay_publisher ReplayPublisher.cpp)
target_compile_features(native_replay_publisher PRIVATE cxx_std_17)
target_include_directories(native_replay_publisher PRIVATE ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(native_replay_publisher rclcpp sensor_msgs)
target_link_libraries(native_replay_publisher ${OpenCV_LIBRARIES})
target_compile_options(native_replay_publisher PRIVATE -Wall -Wextra -Wpedantic)
