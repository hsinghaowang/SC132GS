cmake_minimum_required(VERSION 3.16)
project(euroc_benchmark LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(ov_msckf REQUIRED)
find_package(ov_core REQUIRED)
find_package(ov_init REQUIRED)
find_package(rclcpp REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED)
find_package(Boost REQUIRED COMPONENTS filesystem system thread date_time)
find_package(Ceres REQUIRED)
add_executable(euroc_benchmark main.cpp)
target_compile_features(euroc_benchmark PRIVATE cxx_std_17)
target_compile_definitions(euroc_benchmark PRIVATE ROS_AVAILABLE=2 ENABLE_ARUCO_TAGS=0)
target_include_directories(euroc_benchmark PRIVATE ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(euroc_benchmark ov_msckf ov_core ov_init rclcpp)
target_link_libraries(euroc_benchmark Eigen3::Eigen ${OpenCV_LIBRARIES} ${Boost_LIBRARIES} Ceres::ceres)
