cmake_minimum_required(VERSION 3.20)
project(gs130wi_ros LANGUAGES C CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
    "msg/ImuSample.msg" "msg/ImuBatch.msg" "msg/StereoMetadata.msg"
    DEPENDENCIES std_msgs geometry_msgs)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../gs130wi_capture" "${CMAKE_CURRENT_BINARY_DIR}/capture")
add_executable(gs130wi_publisher src/Publisher.cpp)
set_target_properties(gs130wi_publisher PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_compile_options(gs130wi_publisher PRIVATE -Wall -Wextra -Wpedantic -Werror)
ament_target_dependencies(gs130wi_publisher rclcpp sensor_msgs std_msgs geometry_msgs)
rosidl_get_typesupport_target(typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(gs130wi_publisher gs130wi_capture "${typesupport_target}")
install(TARGETS gs130wi_publisher DESTINATION lib/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
