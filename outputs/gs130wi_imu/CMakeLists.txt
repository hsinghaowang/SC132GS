cmake_minimum_required(VERSION 3.20)
project(gs130wi_imu LANGUAGES CXX)
add_library(icm42688 icm42688.cpp LinuxImuCapture.cpp)
target_compile_features(icm42688 PUBLIC cxx_std_17)
target_include_directories(icm42688 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(icm42688 PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(gs130wi-imu imu_capture.cpp)
target_link_libraries(gs130wi-imu PRIVATE icm42688)
target_compile_options(gs130wi-imu PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(CTest)
if(BUILD_TESTING)
    add_executable(icm42688-check Icm42688Check.cpp)
    target_link_libraries(icm42688-check PRIVATE icm42688)
    target_compile_options(icm42688-check PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME icm42688-fsync COMMAND icm42688-check)
endif()
